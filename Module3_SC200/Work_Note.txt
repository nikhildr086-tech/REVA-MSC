Working Rule:

let timeFrame = 1h;
let failureThreshold = 5;
SigninLogs
| where TimeGenerated >= ago(timeFrame)
| where ResultType !in ("0", "50125")
| summarize 
    failureCount = countif(ResultType != "0"),
    successCount = countif(ResultType == "0"),
    firstFailure = min(TimeGenerated),
    lastSuccess = maxif(TimeGenerated, ResultType == "0")
  by UserPrincipalName, IPAddress, AppDisplayName
| where failureCount >= failureThreshold and successCount > 0
| project UserPrincipalName, IPAddress, AppDisplayName,
         failureCount, successCount, firstFailure, lastSuccess
